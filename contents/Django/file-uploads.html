File Uploads in Django

Accept, store, validate, and serve user-uploaded files and images in Django projects.

📘 How File Uploads Work in Django

When a user submits a file, Django receives it in request.FILES (separate from request.POST). The file is saved to disk at MEDIA_ROOT and accessed via MEDIA_URL.

PartWhat It Does
request.FILESContains uploaded file objects
MEDIA_ROOTDirectory on disk where files are saved
MEDIA_URLURL prefix to access files in browser
FileFieldModel field for any file type
ImageFieldModel field for images (requires Pillow)

⚙️ Step 1 — Configure Settings

# settings.py

import os

MEDIA_URL  = "/media/"
MEDIA_ROOT = os.path.join(BASE_DIR, "media")
# myproject/urls.py  (development only)

from django.conf import settings
from django.conf.urls.static import static

urlpatterns = [
    # ... your patterns
] + static(settings.MEDIA_URL, document_root=settings.MEDIA_ROOT)

🟢 Step 2 — Model with File Fields

# documents/models.py

from django.db import models

class Document(models.Model):
    title       = models.CharField(max_length=200)
    file        = models.FileField(upload_to="documents/")
    uploaded_at = models.DateTimeField(auto_now_add=True)

    def __str__(self):
        return self.title


class Photo(models.Model):
    caption   = models.CharField(max_length=200)
    image     = models.ImageField(upload_to="photos/")
    uploaded_at = models.DateTimeField(auto_now_add=True)

    def __str__(self):
        return self.caption
✔ upload_to="documents/" — files saved to media/documents/filename.pdf.
✔ upload_to can be a function for dynamic paths (e.g. per-user folders).
✔ auto_now_add=True — sets timestamp automatically on creation.

Dynamic upload_to Path

def user_directory(instance, filename):
    return f"users/{instance.user.id}/uploads/{filename}"

class Document(models.Model):
    file = models.FileField(upload_to=user_directory)

🔵 Step 4 — Upload View

# documents/views.py

from django.shortcuts import render, redirect
from .forms import DocumentForm

def upload_document(request):
    form = DocumentForm()

    if request.method == "POST":
        form = DocumentForm(request.POST, request.FILES)   # request.FILES required
        if form.is_valid():
            form.save()
            return redirect("document_list")

    return render(request, "documents/upload.html", {"form": form})
⚠️ Always pass request.FILES as second argument when binding a form with file fields. Without it, the file won't be received even if the user selected one.

📄 Step 5 — Upload Template

<!-- documents/templates/documents/upload.html -->

<h2>Upload Document</h2>
<form method="POST" enctype="multipart/form-data">
  {% csrf_token %}
  {{ form.as_p }}
  <button type="submit">Upload</button>
</form>
⚠️ enctype="multipart/form-data" is mandatory on the <form> tag for file uploads. Without it, request.FILES will always be empty — the file never reaches Django.

🔴 File Validation

Restrict File Type in Form

# documents/forms.py

from django import forms
from .models import Document

class DocumentForm(forms.ModelForm):
    class Meta:
        model  = Document
        fields = ["title", "file"]

    def clean_file(self):
        file = self.cleaned_data.get("file")
        if file:
            allowed = [".pdf", ".docx", ".txt"]
            ext = "." + file.name.split(".")[-1].lower()
            if ext not in allowed:
                raise forms.ValidationError(
                    f"Only {', '.join(allowed)} files are allowed."
                )
            max_size = 5 * 1024 * 1024   # 5 MB
            if file.size > max_size:
                raise forms.ValidationError("File size must be under 5 MB.")
        return file

🗑️ Deleting Files

Deleting a model record does NOT automatically delete the file from disk. Do it manually.

# documents/views.py

import os
from django.shortcuts import get_object_or_404, redirect
from .models import Document

def delete_document(request, pk):
    doc = get_object_or_404(Document, pk=pk)

    if request.method == "POST":
        # Delete file from disk first
        if doc.file and os.path.isfile(doc.file.path):
            os.remove(doc.file.path)
        doc.delete()   # then delete DB record
        return redirect("document_list")

    return render(request, "documents/confirm_delete.html", {"doc": doc})
