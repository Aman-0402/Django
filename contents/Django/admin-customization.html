Django Admin Customization

Transform the default Django admin into a powerful, tailored management interface.

📘 What Is Django Admin?

Django auto-generates a full admin panel at /admin/ for all registered models. Out of the box it works, but customizing ModelAdmin makes it production-ready — searchable, filterable, and much easier to manage.

Access Admin Panel

# 1. Create superuser
python manage.py createsuperuser

# 2. Run server and visit
http://127.0.0.1:8000/admin/

🟢 Basic Registration

# students/admin.py

from django.contrib import admin
from .models import Student

# Simple — shows only __str__ in list
admin.site.register(Student)

🔵 ModelAdmin — Full Customization

# students/admin.py

from django.contrib import admin
from .models import Student

@admin.register(Student)
class StudentAdmin(admin.ModelAdmin):

    # Columns shown in list view
    list_display   = ["id", "name", "age", "email", "course"]

    # Clickable column (opens detail page)
    list_display_links = ["id", "name"]

    # Editable directly in list view (no need to open record)
    list_editable  = ["course"]

    # Sidebar filter panel
    list_filter    = ["course"]

    # Search bar — searches these fields
    search_fields  = ["name", "email"]

    # Default sort order
    ordering       = ["-id"]

    # Records per page
    list_per_page  = 20
✔ @admin.register(Student) decorator replaces admin.site.register(Student, StudentAdmin).
✔ list_editable fields cannot also be in list_display_links.

🟡 Fieldsets — Organize Detail Form

Group fields into sections on the add/edit page.

@admin.register(Student)
class StudentAdmin(admin.ModelAdmin):
    list_display = ["name", "age", "email", "course"]

    fieldsets = [
        ("Personal Info", {
            "fields": ["name", "age", "email"]
        }),
        ("Course Details", {
            "fields": ["course"],
            "classes": ["collapse"],   # collapsible section
        }),
    ]

🔒 Read-Only Fields

@admin.register(Student)
class StudentAdmin(admin.ModelAdmin):
    list_display  = ["name", "email", "course"]
    readonly_fields = ["email"]          # cannot be edited in admin

⚙️ Custom Computed Columns

Add columns that aren't model fields — calculated on the fly.

@admin.register(Student)
class StudentAdmin(admin.ModelAdmin):
    list_display = ["name", "age", "course", "is_adult"]

    @admin.display(description="Adult?", boolean=True)
    def is_adult(self, obj):
        return obj.age >= 18
✔ description — column header text.
✔ boolean=True — shows ✔/✘ icon instead of True/False.

🚀 Custom Admin Actions

Add bulk actions to the dropdown (alongside the default "Delete selected").

from django.contrib import admin, messages

@admin.register(Student)
class StudentAdmin(admin.ModelAdmin):
    list_display = ["name", "course"]
    actions      = ["mark_as_django"]

    @admin.action(description="Move selected students to Django course")
    def mark_as_django(self, request, queryset):
        updated = queryset.update(course="Django")
        self.message_user(
            request,
            f"{updated} student(s) moved to Django course.",
            messages.SUCCESS
        )

🔍 Search Field Lookups

search_fields = [
    "name",           # exact contains (case-insensitive)
    "email",          # contains
    "=course",        # exact match
    "^name",          # starts with
]

🔗 Inline Models — Edit Related Records Together

Show and edit related model records on the parent model's page.

# Example: Course has many Students

from django.contrib import admin
from .models import Course, Student

class StudentInline(admin.TabularInline):   # or StackedInline
    model  = Student
    extra  = 1    # number of empty extra forms shown

@admin.register(Course)
class CourseAdmin(admin.ModelAdmin):
    list_display = ["title", "instructor"]
    inlines      = [StudentInline]
Inline TypeLayout
TabularInlineCompact table — one row per related object
StackedInlineFull form layout — fields stacked vertically

🎨 Site-Wide Admin Customization

# any admin.py (e.g. myproject/admin.py or students/admin.py)

from django.contrib import admin

admin.site.site_header  = "My Project Admin"      # top-left header text
admin.site.site_title   = "My Project"            # browser tab title
admin.site.index_title  = "Welcome to Admin Panel" # dashboard heading

🔄 ManyToMany — filter_horizontal / filter_vertical

Replaces the default multi-select box with a user-friendly dual-list widget.

@admin.register(Course)
class CourseAdmin(admin.ModelAdmin):
    filter_horizontal = ["students"]   # ManyToManyField name
    # OR
    filter_vertical   = ["students"]

📅 Date Hierarchy Navigation

@admin.register(Student)
class StudentAdmin(admin.ModelAdmin):
    list_display   = ["name", "course", "enrolled_on"]
    date_hierarchy = "enrolled_on"    # adds date drill-down nav at top
