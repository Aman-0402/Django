# students/models.py

from django.db import models

class Student(models.Model):
    name = models.CharField(max_length=100)
    age = models.IntegerField()
    email = models.EmailField(unique=True)
    course = models.CharField(max_length=50)

    def __str__(self):
        return self.name

✅ STEP 4: Write View to Retrieve Data

# students/views.py

from django.shortcuts import render
from .models import Student

def student_list(request):
    students = Student.objects.all()   # ORM fetch
    return render(
        request,
        'students/student_list.html',
        {'students': students}
    )
  • Student.objects.all() → fetches all rows
  • students → QuerySet
  • Sent to template using context dictionary

✅ STEP 5: Create Template to Display Data

<!-- students/templates/students/student_list.html -->

<!DOCTYPE html>
<html>
<head>
    <title>Student List</title>
</head>
<body>

<h2>Student Records</h2>

{% if students %}
    <table border="1" cellpadding="8">
        <tr>
            <th>Name</th>
            <th>Age</th>
            <th>Email</th>
            <th>Course</th>
        </tr>

        {% for student in students %}
        <tr>
            <td>{{ student.name }}</td>
            <td>{{ student.age }}</td>
            <td>{{ student.email }}</td>
            <td>{{ student.course }}</td>
        </tr>
        {% endfor %}
    </table>
{% else %}
    <p>No students found.</p>
{% endif %}

</body>
</html>

🔹 Template Language Explanation

Code Meaning
{{ student.name }} Print column value
{% for student in students %} Loop over QuerySet
{% if students %} Check data exists

✅ STEP 6: Connect URL

# students/urls.py

students = Student.objects.order_by('-age')
