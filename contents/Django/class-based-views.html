Class-Based Views (CBV)

Replace repetitive function-based views with Django's powerful built-in generic views.

📘 FBV vs CBV — What's the Difference?

So far every view was a Function-Based View (FBV) — a plain Python function. Class-Based Views (CBV) are Python classes that do the same job but with less code by using Django's built-in generic view classes.

FBVCBV
Plain functionPython class
Write all logic manuallyInherit built-in logic
More code, more controlLess code, faster to write
Good for simple/custom viewsGood for standard CRUD pages
🔑 Both FBV and CBV are valid. Use CBV when doing standard list/detail/create/update/delete. Use FBV for custom logic that doesn't fit a generic pattern.

📂 Project Setup (Same Student Model)

# students/models.py

from django.db import models

class Student(models.Model):
    name   = models.CharField(max_length=100)
    age    = models.IntegerField()
    email  = models.EmailField()
    course = models.CharField(max_length=100)

    def __str__(self):
        return self.name

🟢 ListView — Show All Records

Replaces a view that does Model.objects.all() and renders a template.

# students/views.py

from django.views.generic import ListView
from .models import Student

class StudentListView(ListView):
    model         = Student
    template_name = "students/student_list.html"
    context_object_name = "students"   # name used in template (default: object_list)
# students/urls.py

from django.urls import path
from .views import StudentListView

urlpatterns = [
    path("", StudentListView.as_view(), name="student_list"),
]
<!-- students/templates/students/student_list.html -->

{% for student in students %}
  <p>{{ student.name }} — {{ student.course }}</p>
{% endfor %}
✔ Django automatically queries Student.objects.all().
✔ No need to write the query yourself.
✔ .as_view() converts the class to a callable for urls.py.

🔵 DetailView — Show Single Record

Fetches one record by pk from the URL automatically.

# students/views.py

from django.views.generic import DetailView
from .models import Student

class StudentDetailView(DetailView):
    model         = Student
    template_name = "students/student_detail.html"
    context_object_name = "student"
# students/urls.py

path("<int:pk>/", StudentDetailView.as_view(), name="student_detail"),
<!-- student_detail.html -->

<h2>{{ student.name }}</h2>
<p>Age: {{ student.age }}</p>
<p>Email: {{ student.email }}</p>
<p>Course: {{ student.course }}</p>

🟡 CreateView — Add New Record

Renders a form on GET, validates and saves on POST — automatically.

# students/views.py

from django.views.generic.edit import CreateView
from django.urls import reverse_lazy
from .models import Student

class StudentCreateView(CreateView):
    model         = Student
    fields        = ["name", "age", "email", "course"]
    template_name = "students/student_form.html"
    success_url   = reverse_lazy("student_list")   # redirect after save
# students/urls.py

path("add/", StudentCreateView.as_view(), name="student_create"),
<!-- student_form.html (shared by Create and Update) -->

<form method="POST">
  {% csrf_token %}
  {{ form.as_p }}
  <button type="submit">Save</button>
</form>
✔ Django auto-generates the form from fields.
✔ reverse_lazy() used instead of reverse() because the URL is resolved at class definition time, not runtime.

🟣 UpdateView — Edit Existing Record

Same as CreateView but pre-fills the form with existing data.

# students/views.py

from django.views.generic.edit import UpdateView
from django.urls import reverse_lazy
from .models import Student

class StudentUpdateView(UpdateView):
    model         = Student
    fields        = ["name", "age", "email", "course"]
    template_name = "students/student_form.html"   # same template as Create
    success_url   = reverse_lazy("student_list")
# students/urls.py

path("edit/<int:pk>/", StudentUpdateView.as_view(), name="student_update"),
<!-- Link in list template -->
<a href="{% url 'student_update' student.pk %}">Edit</a>

🔴 DeleteView — Delete a Record

Shows a confirmation page on GET, deletes on POST.

# students/views.py

from django.views.generic.edit import DeleteView
from django.urls import reverse_lazy
from .models import Student

class StudentDeleteView(DeleteView):
    model         = Student
    template_name = "students/student_confirm_delete.html"
    success_url   = reverse_lazy("student_list")
# students/urls.py

path("delete/<int:pk>/", StudentDeleteView.as_view(), name="student_delete"),
<!-- student_confirm_delete.html -->

<p>Delete "{{ student.name }}"?</p>
<form method="POST">
  {% csrf_token %}
  <button type="submit">Yes, Delete</button>
  <a href="{% url 'student_list' %}">Cancel</a>
</form>

📌 All URLs Together

# students/urls.py

from django.urls import path
from .views import (
    StudentListView,
    StudentDetailView,
    StudentCreateView,
    StudentUpdateView,
    StudentDeleteView,
)

urlpatterns = [
    path("",                    StudentListView.as_view(),   name="student_list"),
    path("<int:pk>/",           StudentDetailView.as_view(), name="student_detail"),
    path("add/",                StudentCreateView.as_view(), name="student_create"),
    path("edit/<int:pk>/",      StudentUpdateView.as_view(), name="student_update"),
    path("delete/<int:pk>/",    StudentDeleteView.as_view(), name="student_delete"),
]

⚙️ Customizing CBVs

Override QuerySet (Filter Results)

class StudentListView(ListView):
    model         = Student
    template_name = "students/student_list.html"
    context_object_name = "students"

    def get_queryset(self):
        return Student.objects.filter(course="Django").order_by("name")

Add Extra Context

class StudentListView(ListView):
    model = Student

    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context["total"] = Student.objects.count()
        return context

Override Form (Use ModelForm)

from .forms import StudentModelForm

class StudentCreateView(CreateView):
    model      = Student
    form_class = StudentModelForm       # use custom form instead of fields = [...]
    template_name = "students/student_form.html"
    success_url   = reverse_lazy("student_list")
